Original C Program Implementation (SLA Stage 3)

This is the exact C code model designed for Operating Systems (CMCOR2PC202), implementing both Best-Fit and Worst-Fit algorithms with variable memory partitioning. 

#include <stdio.h>
#define MAX 100

void bestFit(int blocks[], int m, int process[], int n) {
    int i, j, best;
    int freeBlock[MAX];
    for (i = 0; i < m; i++) freeBlock[i] = blocks[i];
    
    printf("\n--- BEST-FIT ALLOCATION ---\n");
    for (i = 0; i < n; i++) {
        best = -1;
        for (j = 0; j < m; j++) {
            if (freeBlock[j] >= process[i] && (best == -1 || freeBlock[j] < freeBlock[best])) {
                best = j;
            }
        }
        if (best != -1) {
            freeBlock[best] -= process[i];
            printf("P%d (%d KB) -> Block %d, Remaining = %d KB\n", i + 1, process[i], best + 1, freeBlock[best]);
        } else {
            printf("P%d (%d KB) -> Not allocated\n", i + 1, process[i]);
        }
    }
}

void worstFit(int blocks[], int m, int process[], int n) {
    int i, j, worst;
    int freeBlock[MAX];
    for (i = 0; i < m; i++) freeBlock[i] = blocks[i];
    
    printf("\n--- WORST-FIT ALLOCATION ---\n");
    for (i = 0; i < n; i++) {
        worst = -1;
        for (j = 0; j < m; j++) {
            if (freeBlock[j] >= process[i] && (worst == -1 || freeBlock[j] > freeBlock[worst])) {
                worst = j;
            }
        }
        if (worst != -1) {
            freeBlock[worst] -= process[i];
            printf("P%d (%d KB) -> Block %d, Remaining = %d KB\n", i + 1, process[i], worst + 1, freeBlock[worst]);
        } else {
            printf("P%d (%d KB) -> Not allocated\n", i + 1, process[i]);
        }
    }
}
